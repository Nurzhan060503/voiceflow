<script>
const SpeechRecognition =
 window.SpeechRecognition || window.webkitSpeechRecognition;

const mic = document.getElementById("mic");
const status = document.getElementById("status");
const result = document.getElementById("result");
const language = document.getElementById("language");
const counter = document.getElementById("counter");

let recognition = null;
let listening = false;

function updateCounter(){
 counter.textContent = result.value.length + " символов";
}
result.addEventListener("input",updateCounter);

function stopListening(){
 if(recognition && listening){
  recognition.stop();
 }
}

mic.addEventListener("click",()=>{
 if(!SpeechRecognition){
  status.textContent="Ваш браузер не поддерживает распознавание речи. Откройте сайт в Chrome или Edge.";
  return;
 }

 if(listening){
  stopListening();
  return;
 }

 recognition = new SpeechRecognition();
 recognition.lang = language.value;
 recognition.continuous = true;
 recognition.interimResults = true;

 recognition.onstart=()=>{
  listening=true;
  mic.classList.add("active");
  mic.textContent="■";
  status.textContent="🎤 Слушаю вас...";
 };

 recognition.onresult=(event)=>{
  let finalPart="";
  let interimPart="";

  for(let i=event.resultIndex;i<event.results.length;i++){
   const transcript=event.results[i][0].transcript;
   if(event.results[i].isFinal){
    finalPart+=transcript+" ";
   }else{
    interimPart+=transcript;
   }
  }

  if(finalPart){
   result.value+=finalPart;
   updateCounter();
  }

  status.textContent=interimPart
   ? "Распознаю: "+interimPart
   : "Текст распознан";
 };

 recognition.onerror=(event)=>{
  status.textContent="Ошибка: "+event.error;
 };

 recognition.onend=()=>{
  listening=false;
  mic.classList.remove("active");
  mic.textContent="🎙";
  if(!status.textContent.startsWith("Ошибка:")){
   status.textContent="Запись остановлена. Нажмите микрофон, чтобы продолжить.";
  }
 };

 try{
  recognition.start();
 }catch(error){
  status.textContent="Не удалось запустить микрофон: "+error.message;
 }
});

language.addEventListener("change",()=>{
 if(listening) stopListening();
 status.textContent="Выбран язык: "+language.options[language.selectedIndex].text;
});

document.getElementById("copy").onclick=async()=>{
 try{
  await navigator.clipboard.writeText(result.value);
  status.textContent="✅ Текст скопирован!";
 }catch(e){
  result.select();
  document.execCommand("copy");
  status.textContent="Текст скопирован!";
 }
};

document.getElementById("download").onclick=()=>{
 const blob=new Blob(["\uFEFF"+result.value],{type:"text/plain;charset=utf-8"});
 const url=URL.createObjectURL(blob);
 const a=document.createElement("a");
 a.href=url;
 a.download="voiceflow-text.txt";
 a.click();
 URL.revokeObjectURL(url);
};

document.getElementById("clear").onclick=()=>{
 result.value="";
 updateCounter();
 status.textContent="Текст очищен";
};

if(!SpeechRecognition){
 status.textContent="Распознавание речи не поддерживается этим браузером.";
}
</script>
